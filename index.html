let players = []; // 記錄上一輪候補 let lastBenchPlayers = []; function addPlayer(){ const name = document.getElementById("playerName") .value .trim(); if(name===""){ alert("請輸入姓名"); return; } if(players.includes(name)){ alert("姓名已存在"); return; } players.push(name); document.getElementById("playerName").value=""; renderPlayers(); } function removePlayer(index){ players.splice(index,1); renderPlayers(); } function clearPlayers(){ players=[]; lastBenchPlayers=[]; renderPlayers(); document.getElementById("result").innerHTML=""; } function renderPlayers(){ let html=""; players.forEach((player,index)=>{ html += ` ${player} `; }); document.getElementById("playerList").innerHTML = html; } function shuffle(array){ for(let i=array.length-1;i>0;i--){ const j = Math.floor( Math.random()*(i+1) ); [array[i],array[j]] = [array[j],array[i]]; } return array; } function generateTeams(){ const courtCount = parseInt( document.getElementById("courtCount").value ); const required = courtCount * 4; if(players.length < required){ alert( `至少需要 ${required} 位球員` ); return; } let playingPlayers = []; let benchPlayers = []; // 人數剛好 if(players.length === required){ playingPlayers = [...players]; shuffle(playingPlayers); lastBenchPlayers = []; } else{ // 上輪候補優先上場 const priorityPlayers = lastBenchPlayers.filter( p => players.includes(p) ); const otherPlayers = players.filter( p => !priorityPlayers.includes(p) ); shuffle(otherPlayers); const orderedPlayers = [ ...priorityPlayers, ...otherPlayers ]; playingPlayers = orderedPlayers.slice( 0, required ); benchPlayers = orderedPlayers.slice( required ); lastBenchPlayers = [...benchPlayers]; } let html = ""; for( let court=1; court<=courtCount; court++ ){ const start = (court-1)*4; const team1 = [ playingPlayers[start], playingPlayers[start+1] ]; const team2 = [ playingPlayers[start+2], playingPlayers[start+3] ]; html += `

${team1[0]}
${team1[1]}
${team2[0]}
${team2[1]}
`; } if(benchPlayers.length > 0){ html += `

🪑 本輪候補名單

${benchPlayers.join("
")}
`; } document.getElementById("result").innerHTML = html; } document .getElementById("playerName") .addEventListener( "keypress", function(e){ if(e.key==="Enter"){ addPlayer(); } } );
